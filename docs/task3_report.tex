\section*{Task description}
{\small
\textbf{Subtasks:} (1)~Load trained TinyLlama LoRA checkpoint. (2)~Extract attention maps from all heads for 20 sample instructions. (3)~Cluster attention heads by cosine similarity. (4)~Token attribution via attention rollout. (5)~HTML heatmaps for 5 detailed + 20 total generations. (6)~Identify \emph{instruction-following} vs \emph{completion} heads empirically. (7)~Compare base model vs LoRA-tuned model quantitatively.

Model: \texttt{TinyLlama-1.1B-Chat-v1.0} + LoRA (\path{adapters/tinyllama-lora-alpaca}, rank~16, $\alpha{=}32$); \textbf{704 total heads} (22 layers $\times$ 32 heads); 20 Alpaca prompts; script: \path{task3_attention_visualization.py}.
}

\section*{Procedure \& technical implementation}
{\small
\begin{enumerate}[label=\textbf{\arabic*.}]
  \item \textbf{Attention extraction.} Monkey-patch \texttt{self\_attn.\_\_call\_\_} to capture softmax weights $(H,L,L)$ per layer in a post-generation full-sequence forward pass (KV-cache calls skipped so only one hook fires per input).
  \item \textbf{Attention rollout.} Cumulative product $R^{(l)} = \mathrm{norm}(A^{(l)}+I)\cdot R^{(l-1)}$ across all 22 layers; output is one attribution vector per sample ranking input tokens by influence on the last output position.
  \item \textbf{Head clustering.} Feature vector per head = concatenated mean instruction- and completion-attention over 20 samples ($\dim = 2\times20$); cosine distance + average-linkage hierarchical clustering (\texttt{scipy}), $k{=}3$ groups, applied independently to base and LoRA.
  \item \textbf{Head classification.} Per head: if mean response$\to$prompt attention $\geq$ mean local-window response attention $\Rightarrow$ \textbf{instruction-following}; else \textbf{completion}. Threshold applied across all 20 prompts.
  \item \textbf{Base vs LoRA comparison.} Layer-wise $\Delta\mathrm{net} = (\mathrm{LoRA\;instr} - \mathrm{LoRA\;comp}) - (\mathrm{Base\;instr} - \mathrm{Base\;comp})$; heatmap and scatter over all 704 heads. Outputs: \path{results/task3/} — 20 HTML heatmaps, 3 clustering PNGs, \path{analysis_summary.json}, \path{layer_analysis.json}, \path{head_classification.json}.
\end{enumerate}
}

\section*{Results}
{\small
\begin{table}[H]
\centering
\caption{Head classification and attention statistics (20 Alpaca prompts, 704 heads). Source: \texttt{analysis\_summary.json}.}
\label{tab:t3}
\footnotesize
\begin{tabular}{@{}lcccc@{}}
\toprule
\textbf{Model} & \textbf{Instruction-following} & \textbf{Completion} & \textbf{Attn.\ focus} & \textbf{Entropy} \\
\midrule
Base  & 445 (63.2\%) & 259 (36.8\%) & 0.5445 & 0.7355 \\
LoRA  & 417 (59.2\%) & 287 (40.8\%) & 0.5999 & 0.7289 \\
$\Delta$ & $-28$ heads ($-4.0\%$) & $+28$ & $+10.2\%$ & $-0.9\%$ \\
\bottomrule
\end{tabular}
\end{table}

\textbf{Key findings.}
\begin{itemize}
  \item \textbf{28 heads} (4\%) shifted instruction$\to$completion after LoRA; all 22 layer $\Delta$net values are negative, confirming a uniform completion-mode amplification across depth.
  \item \textbf{Attention focus +10.2\%} (0.5445$\to$0.5999) with slightly lower entropy: LoRA sharpens prompt-token attribution without dispersing attention.
  \item \textbf{Most affected layers:} 16 ($\Delta{=}{-}0.00301$), 12 ($-0.00248$), 13 ($-0.00243$), 9 ($-0.00236$), 2 ($-0.00208$) --- mid-to-late layers dominate LoRA's recalibration.
  \item \textbf{Stable instruction heads:} L8H11, L15H16, L11H8, L15H20 retain near-identical scores in both models; L1H30 and L14H30 completion heads are amplified in magnitude by LoRA.
  \item \textbf{Cluster structure preserved:} 3-group dendrogram topology identical in base and LoRA; LoRA makes within-cluster refinements only.
\end{itemize}
}
